DATA 	SEGMENT PARA PUBLIC 'DATA'
 
	
DATA 	ENDS

CODE 	SEGMENT PARA PUBLIC 'CODE'

ASSUME CS:CODE, DS:DATA

START	PROC FAR

	PUSH DS
	XOR AX,AX
	PUSH AX	 
	MOV AX,DATA
	MOV DS,AX

	MOV CL,3
	MOV CX,AX
	SHL AX,CL
	SUB AX,CX
	SHL BX,1
	SUB AX,BX
	SHR BX,4
	SUB AX,BX
	

RET		
START	ENDP
CODE 	ENDS
