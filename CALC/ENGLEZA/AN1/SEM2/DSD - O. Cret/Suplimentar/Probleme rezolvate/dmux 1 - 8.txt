entity poarta_si is
	port (a, b:in bit;
	y: out bit);
end poarta_si;	
architecture arh1 of poarta_si is
begin
	y<=a and b;
end arh1;
entity poarta_nu is
	port (a:in bit;
	y: out bit);
end poarta_nu;	
architecture arh2  of poarta_nu is
begin
	y<= not(a);
end arh2;
entity demux12 is
	port(x,s: in bit;
	y1, y0: out bit);
end demux12;
architecture arh of demux12 is
component poarta_si 
	port(a,b:in bit;
	y:out bit);
end component;
component poarta_nu 
	port(a:in bit;
	y:out bit);
end component; 
signal y_temp:bit;
begin
	u1:poarta_si port map(x,s,y1);
	u2:poarta_nu port map(s,y_temp);
	u3:poarta_si port map(x,y_temp,y0);
end arh; 
entity demux18 is
	port(x, s0, s1, s2: in bit;
