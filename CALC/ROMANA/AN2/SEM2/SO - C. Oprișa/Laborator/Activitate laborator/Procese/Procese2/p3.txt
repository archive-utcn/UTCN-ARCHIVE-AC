// 4.  Fie Pi procesul ce verifica daca i este un divizor pentru N dat  si
//determina ordinul de multiplicitate al  acestui  divizor.
//Lansind in executie o structura adecvata de astfel de procese, sa se descompuna
//in factori primi numarul N dat.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <wait.h>
int
main ()
{
  int n;
  int pid;
  int j;
  int m;
  printf ("Dati numarul N=");
  scanf ("%d", &n);
  printf ("Asteptati!!!\n");
  for (int i = 2; i < n; i++)
    {
      pid = fork ();
      if (pid < 0)
	printf ("Eroare la fork!!!\n");
      if (pid == 0)		//suntem in fiu;  
	{
	  m = n;
	  j = 0;
	  printf ("\nsunt procesul %d \n", i);
	  sleep (2);
	  do
	    {
	      if (m % i == 0)
		j = j + 1;
	      m = m / i;
	    }
	  while (m > 0);
	  if (n % i == 0)
	    printf ("\n%d are ordinul de mult %d\n", i, j);
	  return 0;
	}
    }
  printf ("S-a terminat!\n");
}
