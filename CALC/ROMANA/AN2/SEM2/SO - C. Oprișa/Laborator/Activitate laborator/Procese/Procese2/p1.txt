//1. Fie un fisier ce contine in ordine primele N numere 
//naturale. Fie  Pi procesul care elimina toti multiplii 
//lui i  cuprinsi  in intervalul de la 2 la N dat. Folosind 
//un numar adecvat de  astfel de  procese sa se scrie un 
//program care determina toate  numerele prime cuprinse 
//intre 1 si N dat.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <math.h>

int main()
{
FILE *fd;
fd=fopen("input.dat","r");
int sir[100];
int pid;
int i=0,j,n;
int r;
printf("Dati n=");
scanf("%d",&n);
for(i=0;i<n;i++)
 fscanf(fd,"%d",&sir[i]);
i=0;
for(i=2;i<n;i++)
 {
 pid=fork();
 if(pid<0)
  printf("Eroare la FORK!");
 if(pid==0);
  {
  for(j=i;j<n;j++)
   {
   r=sir[j]%i;
   if(r==0)
    sir[j]=0;
   }
  }
 if(pid>0)
  {
  wait(0);
  exit(1);
  }
 }
for(i=1;i<n;i++)
 if(sir[i]!=0)
  printf("Numarul %d este prim!\n",sir[i]);
}
