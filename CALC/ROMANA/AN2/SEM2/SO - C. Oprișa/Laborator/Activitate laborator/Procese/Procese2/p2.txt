//2. Un numar natural este perfect daca este egal
//cu suma divizorilor sai diferiti de el insusi. 
//Folosind un numar adecvat de procese sa se scrie 
//un program care determina toate numerele perfecte 
//mai mici decit N dat.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <wait.h>

int main()
{
int n;
int pid;
int ss=0;
int r;
printf("Dati numarul N=");
scanf("%d",&n);
printf("Asteptati!!!\n");
for(int i=1;i<=n;i++)
 {
 pid=fork();
 if(pid<0)
  printf("Eroare la fork!!!\n");
 if(pid==0) //suntem in fiu;
  {
  for(int j=1;j<i;j++)
   {
   r=i%j;
   if(r==0)
    ss=ss+j;
   }
  if(ss==i)
   printf("%d este numar perfect!\n",i);
  return 0;
  }
 if(pid>0) //suntem in parinte;
  wait(0);
 }
printf("S-a terminat!\n");
}
