void search_files(const char *folder, const char *filename, const char *string) {
    DIR *dir;
    struct dirent *entry;
    struct stat statbuf;
    char path[512];

    dir = opendir(folder);
    if (dir == NULL) {
        perror("Could not open directory");
        return;
    }

    while ((entry = readdir(dir)) != NULL) {
        if (strcmp(entry->d_name, ".") != 0 && strcmp(entry->d_name, "..") != 0) {
            snprintf(path, sizeof(path), "%s/%s", folder, entry->d_name);
            lstat(path, &statbuf);

            if (S_ISDIR(statbuf.st_mode)) {
                // Recursively search subdirectories
                search_files(path, filename, string);
            } else if (S_ISREG(statbuf.st_mode)) {
                if (strcmp(entry->d_name, filename) == 0) {
                    if (fork() == 0) {
                        // Child process
                        execl("/bin/grep", "grep", string, path, NULL);
                        perror("execl failed");
                        exit(1);
                    } else {
                        // Parent process
                        wait(NULL);
                    }
                }
            }
        }
    }

    closedir(dir);
}

int main(int argc, char **argv) {
    if (argc < 4) {
        printf("Usage: %s <folder> <filename> <string>\n", argv[0]);
        return 1;
    }

    const char *folder = argv[1];
    const char *filename = argv[2];
    const char *string = argv[3];

    search_files(folder, filename, string);

    return 0;
}
