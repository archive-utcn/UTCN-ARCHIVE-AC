typedef struct {

    char search_string[MAX_STRING_LENGTH];
} ThreadParams;

void* search_file(void* arg) {
    ThreadParams* params = (ThreadParams*)arg;

    FILE* file = fopen(params->file_path, "r");
    if (file == NULL) {
        perror("Eroare la deschiderea fișierului");
        pthread_exit(NULL);
    }

    char line[MAX_STRING_LENGTH];
    while (fgets(line, sizeof(line), file) != NULL) {
        if (strstr(line, params->search_string) != NULL) {
            printf("%s: %s", params->file_path, line);
            break;
        }
    }

    fclose(file);
    pthread_exit(NULL);
}
