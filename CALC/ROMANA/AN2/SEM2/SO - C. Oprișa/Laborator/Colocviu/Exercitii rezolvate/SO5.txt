/**/#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <string.h>

#define READ_END 0
#define WRITE_END 1

int main(int argc, char* argv[]) {
    // Crearea pipe-urilor
    int pipe_p1_p2[2];
    int pipe_p2_p3[2];

    if (pipe(pipe_p1_p2) == -1 || pipe(pipe_p2_p3) == -1) {
        perror("Eroare la crearea pipe-urilor");
        return 1;
    }

    pid_t p1_pid, p2_pid, p3_pid;

    // Crearea procesului P2
    p1_pid = fork();

    if (p1_pid < 0) {
        perror("Eroare la crearea procesului P2");
        return 1;
    }

    if (p1_pid == 0) {
        // Procesul P2

        close(pipe_p1_p2[WRITE_END]); // P2 nu scrie în pipe_p1_p2

        close(pipe_p2_p3[READ_END]); // P2 nu citește din pipe_p2_p3

        char word[256];
        int vowel_count;

        while (read(pipe_p1_p2[READ_END], word, sizeof(word)) > 0) {
            vowel_count = count_vowels(word);

            write(pipe_p2_p3[WRITE_END], &vowel_count, sizeof(int));
        }

        close(pipe_p1_p2[READ_END]); // Închidem capătul de citire al pipe-ului pipe_p1_p2 în P2

    // Crearea procesului P3

    if (p2_pid < 0) {

    if (p2_pid == 0) {
