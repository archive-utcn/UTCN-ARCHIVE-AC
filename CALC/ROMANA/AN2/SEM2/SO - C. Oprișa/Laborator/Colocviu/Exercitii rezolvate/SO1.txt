#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
/*am umatoarea ierarhie de procese: p1 are ca copil pe p2, p2 are ca copii pe p3 si p4. folosid apelul fork,
implementeaza ierarhia de procese. fiecare proces trebuie sa astepte terminarea propriilor copii. procesul p3
trebuie sa porneasca dupa ce p4 termina.*/

int main(void) {
    pid_t pid1, pid2, pid3, pid4;

    // P1
    pid1 = getpid();
    printf("P1: PID:%d\n", pid1);

    // P2
    pid2 = fork();
    if (pid2 == 0) 
    {
        pid_t ppid = getppid();
        pid2 = getpid();
        printf("P2: PID:%d PPID:%d\n", pid2, ppid);

        // P3
        pid3 = fork();
        if (pid3 == 0) 
        {
            ppid = getppid();
            pid3 = getpid();
            printf("P3: PID:%d PPID:%d\n", pid3, ppid);

            // P4
            pid4 = fork();
            if (pid4 == 0) 
            {
                ppid = getppid();
                pid4 = getpid();
                printf("P4: PID:%d PPID:%d\n", pid4, ppid);
                printf("P4: Terminating\n");
                exit(0);
            } else if (pid4 > 0) 
            {
                // P3 așteaptă terminarea copilului P4 înainte de a se termina
                waitpid(pid4, NULL, 0);
                printf("P3: Terminating\n");
                exit(0);
            } else 
            {
                perror("P4 fork failed");
                exit(1);
            }
        } else if (pid3 > 0) 
        {
            // P2 așteaptă terminarea copilului P3 înainte de a se termina
            waitpid(pid3, NULL, 0);
            printf("P2: Terminating\n");
            exit(0);
        } else 
        {
            perror("P3 fork failed");
            exit(1);
        }
    } else if (pid2 > 0) 
    {
        // P1 așteaptă terminarea copilului P2
        waitpid(pid2, NULL, 0);
        printf("P1: Terminating\n");
    } else 
    {
        perror("P2 fork failed");
        exit(1);
    }

    return 0;
}
