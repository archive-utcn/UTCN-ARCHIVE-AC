public abstract class  MedicalItem{
	
	private int id;
	private String nume;
	private String CNP;
	
	public MedicalItem(int id, String nume,String CNP)
	{
		this.id = id;
		this.nume = nume;
		this.CNP = CNP;
	}
	
	public MedicalItem(){}
	
	public abstract void print();
	
	public int getId(){
		return id;
	}
	
	public void setId (int id){
		this.id = id;
	}
	
	public String getNume(){
		return nume;
	}
	
	public void setNume (String nume){
		this.nume = nume;
	}
	
	public String getCNP(){
		return CNP;
	}
	
	public void setCNP (String CNP){
		this.CNP = CNP;
	}
}

public class  MedicalNote extends MedicalItem {
	
	private String descriere;
	private String nume;
	
	public MedicalNote(String descirere, String nume){
		this.descriere = descriere;
		this.nume = nume;
	}
	
	public MedicalNote(){}
	
	@overide
	public void print(){}
	
	public String getNume(){
		return nume;
	}
	
	public void setNume (String nume){
		this.nume = nume;
	}
	
	public String getDescriere(){
		return descriere;
	}
	
	public void setDescriere (String descriere){
		this.descriere = descriere;
	}	
}
