--1.a. member
member e xs = foldr (||) False (map ((==) e) xs)

--1.b. true if all the elements from a list satisfy predicate p
alll p xs = foldr (||) False (map p xs)

--2.a. map function on the tree data structure
data Tree = Leaf Int | Node Tree Tree deriving Show

sumTree (Leaf x) = x
sumTree (Node l r) = sumTree l + sumTree r 

mapTree (f, (Leaf x)) = (Leaf (f x))
mapTree (f, (Node l r)) = (Node (mapTree (f, l)) (mapTree (f, r)))

--2.b. zero all negative numbers from a tree

zeroOut = mapTree ((\ x -> if (x<0) then 0 else x), (Node (Leaf (-1)) (Leaf 6)))

--3.a. f::String->Bool to verify that every vowel in a string is uppercase

andA l = (length [x | x<-l, x == False ]) == 0
isUpperA c = member c ['A'..'Z']
isVowelA c = member c ['a','e','i','o','u', 'A','E','I','O','U']
f l = andA [isUpperA x | x <- l, isVowelA x]

--3.b. g::String->Bool like f, using basic functions and recursion, but not list comprehension or library functions 

andAl [] = True
andAl [x] = x
andAl (x:xs) = if x==False then False else andAl xs

check [] l = []
check (x:xs) l | member x l = False:check xs l
 | otherwise = check xs l
 
g l = andAl (check l ['a','e','i','o','u'])
