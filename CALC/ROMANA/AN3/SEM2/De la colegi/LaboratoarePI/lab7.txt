Mat dilate(Mat src)
{
	int height = src.rows;
	int width = src.cols;
	int i = 0;
	int j = 0;
	int auxx = 0;
	int auxy = 0;
	int x = 0;
	int y = 0;
	int k = 0;
	
	Mat structural(3, 3, CV_8UC1, Scalar(255));
	Mat dst(height, width, CV_8UC1, Scalar(255));

	structural.at<uchar>(0, 1) = 0;
	structural.at<uchar>(1, 1) = 0;
	structural.at<uchar>(1, 0) = 0;
	structural.at<uchar>(1, 2) = 0;
	structural.at<uchar>(2, 1) = 0;

		for (i = 1; i < height - 1; i++)
		{
			for (j = 1; j < width - 1; j++)
			{
				if (src.at<uchar>(i, j) == 0)
				{
					for (x = 0; x < 3; x++)
					{
						for (y = 0; y < 3; y++)
						{
							if (structural.at<uchar>(x, y) == 0) {
								auxx = i + x - (structural.rows / 2);
								auxy = j + y - (structural.cols / 2);
								dst.at<uchar>(auxx, auxy) = 0;
							}

						}
					}
				}
			}
		}

		return dst;
}

Mat open(Mat src) {

	Mat dst = src.clone();

	int height = src.rows;
	int width = src.cols;
	int i = 0;
	int j = 0;
	int x = 0;
	int y = 0;
	int auxx = 0;
	int auxy = 0;

	Mat structural(3, 3, CV_8UC1, Scalar(255));
	structural.at<uchar>(0, 1) = 0;
	structural.at<uchar>(1, 1) = 0;
	structural.at<uchar>(1, 0) = 0;
	structural.at<uchar>(1, 2) = 0;
	structural.at<uchar>(2, 1) = 0;
	
	for ( i = 1; i < height - 1; i++) 
	{
		for ( j = 1; j < width - 1; j++)
		{
			if (src.at<uchar>(i, j) == 0) 
			{
				for ( x = 0; x < 3; x++) {
					for ( y = 0; y < 3; y++)
					{
						auxx = i + x - (structural.rows / 2);
						auxy = j + y - (structural.cols / 2);
						if (structural.at<uchar>(x, y) == 0 && src.at<uchar>(auxx,auxy ) == 255) 
						{
							dst.at<uchar>(i, j) = 255;
						}

					}
				}
			}
		}
	}

	Mat dstcpy = dst.clone();

	for ( i = 1; i < height - 1; i++) 
	{
		for ( j = 1; j < width - 1; j++) 
		{
			if (dstcpy.at<uchar>(i, j) == 0) 
			{
				for ( x = 0; x < 3; x++) {
					for ( y = 0; y < 3; y++)
					{
						if (structural.at<uchar>(x, y) == 0) 
						{
							auxx = i + x - (structural.rows / 2);
							auxy = j + y - (structural.cols / 2);
							dst.at<uchar>(auxx,auxy ) = structural.at<uchar>(x, y);
						}

					}
				}
			}
		}
	}

	return dst;

}

void contur() {
	char fname[MAX_PATH];
	openFileDlg(fname);
	Mat src = imread(fname, IMREAD_GRAYSCALE);
	Mat dst = src.clone();

	int height = src.rows;
	int width = src.cols;
	int i = 0;
	int j = 0;
	int x = 0;
	int y = 0;
	int auxx = 0;
	int auxy = 0;

	Mat structural(3, 3, CV_8UC1, Scalar(255));
	structural.at<uchar>(0, 1) = 0;
	structural.at<uchar>(1, 1) = 0;
	structural.at<uchar>(1, 0) = 0;
	structural.at<uchar>(1, 2) = 0;
	structural.at<uchar>(2, 1) = 0;

	imshow("IMG", src);

	for (i = 1; i < height - 1; i++)
	{
		for (j = 1; j < width - 1; j++) 
		{
			if (src.at<uchar>(i, j) == 0)
			{
				for ( x = 0; x < 3; x++)
				{
					for ( y = 0; y < 3; y++)
					{
						auxx = i + x - (structural.rows / 2);
						auxy = j + y - (structural.cols / 2);
						if (structural.at<uchar>(x, y) == 0 && src.at<uchar>(auxx,auxy ) == 255) {
							dst.at<uchar>(i, j) = 255;
						}

					}
				}
			}
		}
	}

	Mat dstcpy = dst.clone();
	dst = src.clone();

	for (i = 1; i < height - 1; i++) 
	{
		for (j = 1; j < width - 1; j++)
		{
			if (src.at<uchar>(i, j) == dstcpy.at<uchar>(i, j))
			{
				dst.at<uchar>(i, j) = 255;
			}
			else dst.at<uchar>(i, j) = src.at<uchar>(i, j);
		}
	}
	imshow("Contur", dst);
	waitKey();
}
