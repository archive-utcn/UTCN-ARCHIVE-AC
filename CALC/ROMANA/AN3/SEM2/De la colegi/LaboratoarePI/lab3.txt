void hist()
{
	char fname[MAX_PATH];
	int h[256];
	int i = 0;
	int j = 0;

	while (openFileDlg(fname))
	{
		Mat_<uchar> img = imread(fname, IMREAD_GRAYSCALE);
		int rows = img.rows;
		int cols = img.cols;
		uchar v;

		for (i = 0; i < 256; i++)
		{
			h[i] = 0;
		}

		for (i = 0; i < rows; i++)
		{
			for (j = 0; j < cols; j++)
			{
				v = img.at<uchar>(i, j);
				h[v]++;
			}
		}

		imshow("imagine", img);
		showHistogram("histogram", h, 255, 200);
		waitKey(0);
	}

}

	int j = 0;

	float v = 0.00f;
	float sum = 0.0f;
	bool ok = true;
	
	vf.push_back(0);

	while (openFileDlg(fname))
	{
		Mat img = imread(fname, IMREAD_GRAYSCALE);
		int rows = img.rows;
		int cols = img.cols;

		for (i = 0; i < 256; i++)
		{
			h[i] = 0;
		}

		for (i = 0; i < rows; i++)
		{
			for (j = 0; j < cols; j++)
			{
				uchar v1 = img.at<uchar>(i, j);
				h[v1]++;
			}
		}

		for (int i = 0; i < 256; i++)
		{
			fdp[i] = (float)h[i] / rows * cols;
			//std::cout << h[i] << " ";
		}

		showHistogram("pas1+2", h, 255, 200);
	
		for ( k = WH; k < 255 - WH; k++) 
		{
			
			
			for (int i = k - WH; i <= k + WH; i++) 
			{
				sum += fdp[i];
				if ((i != k) && (fdp[i] > fdp[k]))
				{
					ok = false;
				}
			}

			 v = sum / (float)(2 * WH + 1);

			if ((fdp[k] > v + TH) && ok)
			{
				vf.push_back(k);
			}
		}

		vf.push_back(255);

		for ( i = 0; i < 256; i++) 
		{
			fdp2[i] = 0;
		}

		int peaksLength = vf.size();

		for ( i = 0; i < peaksLength; i++) 
		{
			fdp2[vf[i]] = h[vf[i]];
			printf(" %d ", fdp2[vf[i]]);
		}
