void inversa() 
{
	Mat_<float> A(3, 3);
	Mat_<float> transpusa(3, 3);
	Mat_<float> inv(3, 3);
	float det = 0;
	int i = 0;
	int j = 0;

	for (i = 0; i < 3; i++)
	{
		for (j = 0; j < 3; j++)
		{
			A(i, j) = 4 - j;
		}
	}

	det = A(0, 0) * A(1, 1) * A(2, 2) +
		A(1, 0) * A(2, 1) * A(0, 2) +
		A(0, 1) * A(1, 2) * A(2, 0) -
		A(0, 2) * A(1, 1) * A(2, 0) -
		A(0, 1) * A(1, 0) * A(2, 2) -
		A(1, 2) * A(2, 1) * A(0, 0);

	for (i = 0; i < 3; i++)
	{
		for (j = 0; j < 3; j++)
		{
			transpusa(i, j) = A(j, i);

		}
	}

	inv(0, 0) = A(1, 1) * A(2, 2) - A(1, 2) * A(2, 1);
	inv(0, 2) = A(0, 1) * A(1, 2) - A(1, 1) * A(0, 2);
	inv(1, 1) = A(0, 0) * A(2, 2) - A(0, 2) * A(2, 0);
	inv(2, 0) = A(1, 0) * A(2, 1) - A(2, 0) * A(1, 1);
	inv(2, 2) = A(1, 1) * A(0, 0) - A(1, 0) * A(0, 1);

	inv(0, 1) = (-1) * (A(0, 1) * A(2, 2) - A(2, 1) * A(0, 2));
	inv(1, 0) = (-1) * (A(1, 0) * A(2, 2) - A(1, 2) * A(2, 0));
	inv(1, 2) = (-1) * (A(0, 0) * A(1, 2) - A(1, 0) * A(0, 2));
	inv(2, 1) = (-1) * (A(0, 0) * A(2, 1) - A(0, 1) * A(2, 0));

	for (i = 0; i < 3; i++) {

		for (j = 0; j < 3; j++)
		{
			inv(i, j) = inv(i, j) / det;
		}
	}

	for (i = 0; i < 3; i++)
	{
		for (j = 0; j < 3; j++)
		{
			std::cout << inv(i, j);
		}

		std::cout << "\n";
	}

	{
		Mat_<Vec3b> img = imread(fname, IMREAD_GRAYSCALE);

		Vec3b pixel;
		Mat_<uchar> red(rows, cols);
		Mat_<uchar> blue(rows, cols);
		Mat_<uchar> green(rows, cols);
		
		for ( i = 0; i < rows; i++) 
		{
			for ( j = 0; j < cols; j++) 
			{
				pixel[0] = img(i, j)[0];
				pixel[1] = pixel[2] = 0;
				blue.at<Vec3b>(i, j) = pixel;

				pixel[1] = img(i, j)[1];
				pixel[0] = pixel[2] = 0;
				green.at<Vec3b>(i, j) = pixel;

				pixel[2] = img(i, j)[2];
				pixel[0] = pixel[1] = 0;
				red.at<Vec3b>(i, j) = pixel;
			}
		}

		imshow("blue", blue);
		imshow("green", green);
		imshow("red", red);
		waitKey(0);
	}

}

EX4 - CONVERSIE RGB->HSV


void RGBtoHSVconversion() {
	Mat imgOriginal = imread("Images/flowers_24bits.bmp", 1);
	Mat img = imgOriginal.clone(); //am clonat imaginea ca sa nu se lucreze pe imaginea originala
	Mat imgH(img.rows, img.cols, CV_8UC1); //noile imagini pe care le vom crea in functie de H, S, V normalizate vor avea un singur canal: CV_8UC1
	Mat imgS(img.rows, img.cols, CV_8UC1);
	Mat imgV(img.rows, img.cols, CV_8UC1);
	float r, g, b, M, m, C, V, S, H, H_norm, S_norm, V_norm;

	for (int i = 0; i < img.rows; i++) {
		for (int j = 0; j < img.cols; j++) {
			Vec3b pixel = img.at< Vec3b>(i, j);
			unsigned char B = pixel[0];
			unsigned char G = pixel[1];
			unsigned char R = pixel[2];
			
			r = (float) R / 255;
			g = (float) G / 255;
			b = (float) B / 255;

			M = max3float(r, g, b);
			m = min3float(r, g, b);
			
			C = M - m;
			V = M;
			if (V != 0) {
				S = C / V;
			}
			else S = 0; //avem de a face cu o imagine gray scale

			if (C != 0) {
				if (M == r) {
					H = 60 * (g - b) / C;
				}
				if (M == g) {
					H = 120 + 60 * (b - r) / C;
				}
				if (M == b) {
					H = 240 + 60 * (r - g) / C;
				}
			}
			else H = 0;
			
			if (H < 0) {
				H = H + 360;
			}

			H_norm = H * 255 / 360;
			S_norm = S * 255;
			V_norm = V * 255;

			imgH.at<unsigned char>(i, j) = H_norm; 
			imgS.at<unsigned char>(i, j) = S_norm;
			imgV.at<unsigned char>(i, j) = V_norm;

		}
	}

	imshow("Original image", imgOriginal); //se va afisa initial imaginea originala, pe urma cele trei imagini corespunzatoare H, S, V
	waitKey(0);

	imshow("H image", imgH);

	imshow("S image", imgS);

	imshow("V image", imgV);
	waitKey(0);
}
