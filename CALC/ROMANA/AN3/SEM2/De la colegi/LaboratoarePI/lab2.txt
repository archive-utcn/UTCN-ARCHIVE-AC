void canaleRGB() 
{
	char fname[MAX_PATH];
	int i = 0;
	int j = 0;
	while (openFileDlg(fname))
	{
		Mat_<Vec3b> img = imread(fname, IMREAD_GRAYSCALE);
		int rows = img.rows;
		int cols = img.cols;
		Vec3b pixel;
		Mat_<uchar> red(rows, cols);
		Mat_<uchar> blue(rows, cols);
		Mat_<uchar> green(rows, cols);
		
		for ( i = 0; i < rows; i++) 
		{
			for ( j = 0; j < cols; j++) 
			{
				pixel[0] = img(i, j)[0];
				pixel[1] = pixel[2] = 0;
				blue.at<Vec3b>(i, j) = pixel;

				pixel[1] = img(i, j)[1];
				pixel[0] = pixel[2] = 0;
				green.at<Vec3b>(i, j) = pixel;

				pixel[2] = img(i, j)[2];
				pixel[0] = pixel[1] = 0;
				red.at<Vec3b>(i, j) = pixel;
			}
		}

		imshow("blue", blue);
		imshow("green", green);
		imshow("red", red);
		waitKey(0);
	}

}

void grayscale()
{
	char fname[MAX_PATH];
	int i = 0;
	int j = 0;

	while (openFileDlg(fname))
	{
		Mat_<Vec3b> img = imread(fname, IMREAD_COLOR);
		Mat_<uchar> imggray(img.rows, img.cols);
		int rows = img.rows;
		int cols = img.cols;
		for (i = 0; i < rows; i++)
		{
			for (j = 0; j < cols; j++)
			{
				imggray(i, j) = (img(i, j)[0] + img(i, j)[1] + img(i, j)[2]) / 3;
			}
		}

		imshow("poza netransformata", img);
		imshow("grayscale", imggray);
		waitKey(0);
	}
}


void binar(int prag) {

	char fname[MAX_PATH];
	int i = 0;
	int j = 0;
	
	while (openFileDlg(fname))
	{
		Mat_<uchar> img = imread(fname, IMREAD_GRAYSCALE);
		int rows = img.rows;
		int cols = img.cols;
		Mat_<uchar> dst(rows, cols);
		for (i = 0; i < rows; i++)
		{
			for (j = 0; j < cols; j++)
			{
				if (img(i, j) < prag) {
					dst(i, j) = 0;
				}
				else if (img(i, j) > prag) {
					dst(i, j) = 255;
				}
			}
		}
		imshow("inainte", img);
		imshow("dupa", dst);
		waitKey(0);
	}

	
}

void convHSV() {

	char fname[MAX_PATH];

	while (openFileDlg(fname))
	{

		Mat_<Vec3b> img = imread(fname, IMREAD_COLOR);
		Mat_<Vec3b> rgb(img.rows, img.cols);

		Mat_<Vec3b> convH(img.rows, img.cols);
		Mat_<Vec3b> convS(img.rows, img.cols);
		Mat_<Vec3b> convV(img.rows, img.cols);

		float r, g, b, m, M, C, H, S, V;
		float H_norm, S_norm, V_norm;
		int i = 0;
		int j = 0;
		int rows = img.rows;
		int cols = img.cols;

		for (i = 0; i < rows; i++)
		{
			for (j = 0; j < cols; j++)
			{
				r = img(i, j)[2] / 255.0;
				g = img(i, j)[1] / 255.0;
				b = img(i, j)[0] / 255.0;

				M = max(max(r, g), b);
				m = min(min(r, g), b);
				C = M - m;
				V = M;

				if (V != 0) {
					S = C / V;
				}
				else S = 0;

				if (C != 0)
				{
					if (M == r)
					{
						H = 60 * (g - b) / C;
					}
					if (M == g)
					{
						H = 120 + 60 * (b - r) / C;
					}
					if (M == b)
					{
						H = 240 + 60 * (r - g) / C;
					}
				}
				else
				{

					H = 0;
				}
				if (H < 0)
				{
					H = H + 360;
				}

				H_norm = H * 255 / 360;
				S_norm = S * 255;
				V_norm = V * 255;

				convH(i, j)[0] = H;
				convS(i, j)[1] = S_norm;
				convV(i, j)[2] = V_norm;

			}

		}

		imshow("H", convH);
		imshow("S", convS);
		imshow("V", convV);
		waitKey(0);
	}
}

bool isInside(Mat img, int i, int j)
{
	int rows = img.rows;
	int cols = img.cols;
	if (i >= 0 && i < rows && j >= 0 && j < cols)
