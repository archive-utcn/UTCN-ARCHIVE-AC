int pal(int n, int aux){
	if (n == 0 && aux == 0)
		return 1;
	if (n%10 == 0 && aux == 0)
		return 0;


	if (aux > n)
		return 0;
	if (aux == n || aux == n / 10)
		return 1;

	return pal(n/10, aux*10 + n%10);

}
