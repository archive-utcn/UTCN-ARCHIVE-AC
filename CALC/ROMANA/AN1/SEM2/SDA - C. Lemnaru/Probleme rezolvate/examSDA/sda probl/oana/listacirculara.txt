typedef struct nod
{
    int info;
    struct nod *urm;
}Nod;

void adauga(Nod **prim,Nod **ultim,int nr)
{
    Nod *p;
    p=(struct nod *)malloc(sizeof(struct nod));
    if(*prim==0)
      {
        (*prim)=(*ultim)=p;
        (*prim)->info=nr;
      }
     else
      {
        (*ultim)->urm=p;
        (*ultim)=p;
        (*ultim)->info=nr;
      }
     (*ultim)->urm=0;
}
