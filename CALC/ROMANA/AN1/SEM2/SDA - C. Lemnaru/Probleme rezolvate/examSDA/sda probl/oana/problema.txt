struct nod{
        char cuv[20];
        struct nod *urm;
        };
struct nod *prim,*ultim,*p,*q;
int creare(char sir[])
{
        if(p==NULL)
    {
        p=(struct nod *)malloc(sizeof(struct nod));
         if(p==NULL)
         {
             printf("RAM plin!!!");
             exit(0);
         }
        prim=p;
        ultim=p;
        strcpy(p->cuv,sir);
        ultim->urm=0;
    }

    else
    {

        p=(struct nod *)malloc(sizeof(struct nod));
        if(p==NULL)
         {
             printf("RAM plin!!!");
             exit(0);
         }
        ultim->urm=p;
        ultim=p;
        strcpy(p->cuv,sir);
        ultim->urm=0;
    }
    return (p);
}
void afisare()
{
    p=prim;
    while(p!=NULL)
    {
        printf("%5s",p->cuv);
        p=p->urm;
    }
}
int main()
{
    int n,i;
    char sir[20];
    struct nod *p=NULL;
    printf("dati val lui n");
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        scanf("%s",&sir);
        p=creare(sir);
