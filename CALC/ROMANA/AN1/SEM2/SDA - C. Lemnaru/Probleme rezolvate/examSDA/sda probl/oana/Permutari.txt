#include <stdio.h>
#include <stdlib.h>

int x[20],n=3;
void tipar(int k)
    {
        int i;
        for (i=1;i<=k;i++)
        printf("%d ",x[i]);
        printf("\n");

    }

void backtrp(int i)
    {
        int j,p,ok;
        for (j=1;j<=n;j++)
            {
                x[i]=j;
                ok=1;
                for (p=1;p<i && ok==1;p++)
                    if ( x[i]==x[p] ) ok=0;

            if (ok)
                if (i==n) tipar (n);
                else backtrp(i+1);
            }
    }

int main()
{
    backtrp(1);
    return 0;
}
