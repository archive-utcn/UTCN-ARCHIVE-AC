#include <stdio.h>
#include <stdlib.h>

int x[20],n=5,k=3;
void tipar (int nr)
    {
        int i;
        for (i=1;i<=nr;i++)
            printf("%d ",x[i]);
            printf("\n");
    }
void backtr(int i)
    {
        int j;
        for (j=1;j<=n;j++)
        {

         x[i]=j;
        if ( (i==1) || (x[i]>x[i-1]) )
         if (i==k)
            tipar (k);
        else backtr(i+1);
        }
    }

int main()
{
  backtr(1);
    return 0;
}
