 void shell_sort(int n,float a[nmax])
 /* SORTAREA PRIN METODA SHELL */
 {
	int i,j,incr;
	float x;
	incr=1;
	while(incr<n)
		incr=incr*3 + 1;
	while(incr>=1)
	  {
		 incr=incr / 3;
		 for(i=incr;i<n;i++)
		       {
			  x=a[i];j=i;
			  while(a[j-incr]>x)
				 {
					a[j]=a[j-incr];
					j=j-incr;
					if(j<incr) break;
				  };
			    a[j]=x;
		         }
	    }
 }
