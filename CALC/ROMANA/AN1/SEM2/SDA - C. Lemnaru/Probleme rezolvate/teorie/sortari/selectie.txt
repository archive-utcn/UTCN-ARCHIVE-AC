
    void sort_selectie(int n,float a[nmax])
    /* SORTAREA PRIN SELECTIE */
    {
	  int i,j,poz;
	  float x;
	  for(i=0;i<n-1;i++)
	         {
		  x=a[i];poz=i;
		  for(j=i+1;j<n;j++)
		  if(a[j]<x) {
			          x=a[j];
			          poz=j;
			       }
		  a[poz]=a[i];a[i]=x;
		}
         }
