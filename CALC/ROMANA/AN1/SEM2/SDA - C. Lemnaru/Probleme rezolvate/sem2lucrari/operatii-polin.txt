#include <stdio.h>
#include <stdlib.h>

typedef struct nod
{
    float coef;
    int grad;
    struct nod *urm;
}polinom;
polinom *x,*y,*aux, *scade;

void adaugare(float coef, int grad, polinom **a)
{
    polinom *p=(polinom*)malloc(sizeof(polinom));
    polinom *k;
    p->coef=coef;
    p->grad=grad;
    p->urm=0;
    if (*a==0)  *a=p;
    else
    {
        k=*a;
        while (k->urm!=0)  k=k->urm;
        k->urm=p;
    }
}

void afisare (polinom **a )
{
    polinom *k=*a;
    int i=0;
    while (k!=0)
    {
        if (i==0) ;
        if (k->coef > 0)  printf("+%0.1fx^%d",k->coef,k->grad);

        else if (k->coef < 0)  printf("%0.1fx^%d",k->coef,k->grad);
        else ;

        k=k->urm;
        i++;
    }

}

void adunare()
{
    polinom *p=x;
    polinom *q=y;
    polinom *suma=0;


    while (p!=0 && q!=0)
    {
        if (p->grad==q->grad)
        {
            adaugare(p->coef+q->coef,p->grad,&suma);
            p=p->urm;
            q=q->urm;
        }
        else if (p->grad>q->grad)
        {
            adaugare(p->coef,p->grad,&suma);
            p=p->urm;
        }
        else
        {
            adaugare(q->coef,q->grad,&suma);
            q=q->urm;
        }
    }
    if (p!=0)  while (p!=0)
        {
            adaugare(p->coef,p->grad,&suma);
            p=p->urm;
        }
    if (q!=0)  while (q!=0)
        {
            adaugare(q->coef,q->grad,&suma);
            q=q->urm;
        }
    printf("suma=");
    afisare(&suma);
}

void scadere(polinom **a,polinom **b)
{
    polinom *p=*a;
    polinom *q=*b;
    scade=0;


    while (p!=0 && q!=0)
    {
        if (p->grad==q->grad)
        {
            adaugare(p->coef-q->coef,p->grad,&scade);
            p=p->urm;
            q=q->urm;
        }
        else if (p->grad>q->grad)
        {
            adaugare(p->coef,p->grad,&scade);
            p=p->urm;
        }
        else
        {
            adaugare(q->coef*(-1),q->grad,&scade);
            q=q->urm;
        }
    }
    if (p!=0)  while (p!=0)
        {
            adaugare(p->coef,p->grad,&scade);
            p=p->urm;
        }
    if (q!=0)  while (q!=0)
        {
            adaugare(q->coef*(-1),q->grad,&scade);
            q=q->urm;
        }

}

void inmultire(polinom **a, polinom **b)
{
    polinom *p=*a;
    polinom *q=*b;
    aux=0;
    while (p!=0)
    {
        while (q->urm!=0)
        {
            adaugare(p->coef * q->coef, p->grad+q->grad, &aux);
            q=q->urm;
        }
        q=*b;
        p=p->urm;
    }

    p=aux;
    polinom *k;
    while (p!=0)
    {
        k=p;
        while (k->urm!=0)
        {
            if (k->urm->grad==p->grad)
            {
                p->coef+=k->urm->coef;
                q=k->urm;
                k->urm=q->urm;
                free(q);
            }
            k=k->urm;
        }
        p=p->urm;
    }


}

{
        polinom *p=*a;
        polinom *q=*b;
        polinom *cat=0;
        while(p->grad >= q->grad)
        {
                adaugare(p->coef/q->coef,p->grad-q->grad,&cat);
                polinom *k=cat;
                while(k->urm!=0) k=k->urm;
                inmultire(&k,&q);
                scadere(&p,&aux);

        }
        printf("\n cat=");

}
