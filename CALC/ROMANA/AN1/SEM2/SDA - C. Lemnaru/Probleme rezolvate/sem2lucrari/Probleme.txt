#include "conio.h"

typedef struct arc { int v1, v2;
                     int cost;
                     arc *ant,*nx;
                   };

typedef struct antseledist { int id;
                             antseledist *nx,*ant,*ptr;
                           };

void adauganodsol(antseledist *ante, int val, int cod)   // cod==0  recreare solutie
{ antseledist *p,*aux=ante->ptr;                         // altfel adaugare normala
  if (!cod)
     { while (aux)
             { p=aux;
               aux=aux->nx;
               free(p);
             }
       aux=new antseledist;
       aux->id=val;
       aux->ptr=aux->ant=ante;
       aux->nx=NULL;
       ante->ptr=aux;
       ante->id=1;
       return;
     }
  p=new antseledist;
  p->id=val;
  p->nx=ante->ptr;
  if (p->nx)
     p->nx->ant=p;
  p->ptr=ante;
  p->ant=ante;
  ante->ptr=p;
  ante->id++;
}

void adaugaasd(antseledist **a,int val)
{ antseledist *p=new antseledist;
  p->id=val;
  p->nx=p->ant=NULL;
  if ( !(*a) )
     { (*a)=p;
       return;
     }
  p->nx=(*a);
  (*a)->ant=p;
  (*a)=p;
}


// adaugare de muchie in lista muchiilor astfel incat lista sa fie ordonata
// se foloseste o lista dublu inlantuita
void adaugaordonat(arc **a,int v1, int v2, int cost)
{ arc *p=new arc;
  arc *au=*a;
  p->v1=v1;
  p->v2=v2;
  p->cost=cost;         // se completeaza campul costului
  p->nx=p->ant=NULL;    // acesta inca nu are succesor sau predecesor
  if ( !(*a) )
     { *a=p;            // daca se adauga primul elelement
       return;          // acesta este chiar p, si se iese din functie
     }
  if ( (p->v2==(*a)->v2) && (p->v1==(*a)->v1) )
     { printf("Nod existent");    //nodul de adaugat se afla in lista deja, pe prima pozitzie
       return;                    // se iese din functzie
     }
  if ( (p->v1<(*a)->v1) || ( (p->v1==(*a)->v1) && (p->v2<(*a)->v2) ) )   // daca varful1 de adaugat este mai mic decat primul sau daca varful1 de adaugat este egal cu primul si daca varful2 de adaugat este mai mic decat varful2 atunci
     { p->nx=(*a);           // noua muchie se adauga la inceput
       (*a)->ant=p;
       (*a)=p;
       return;        // si se iese din functie
     }
  while ( (au->nx) && (p->v1>au->nx->v1) )   // se avanseaza in lista pana cand varful1 muchiei de adaugat este egal cu varful1 urmator elementului curent din lista si cat timp nu s-a terminat lista
        au=au->nx;
  while ( (au->nx) && (p->v1==au->nx->v1) && (p->v2>au->nx->v2) )  // se avanseaza in lista pana cand varful2 urmator elementului curent din lista este mai mare decat varful2 de adaugat si cat timp varful1 de adaugat este egal cu varfurile din lista si cat timp nu s-a terminat lista
        au=au->nx;
  if ( (p->v2==au->nx->v2) && (p->v1==au->nx->v1) )     // daca s-a dat peste un nod existent
     { printf("Nod existent");                          // mesaj corespunzator
       return;                                          // si se iese din functie
     }
  p->nx=au->nx;         // adaugare dupa elementul curent
  p->ant=au;            // legaturile de rigoare intr-o lista dublu inlantzuita
  if (au->nx)           // daca se face adaugare la sfarsit trebuie avut in vedere ca nu exista element urmator celui adaugat acuma
     au->nx->ant=p;
  au->nx=p;
  return;               // si se iese din functie
}

antseledist* cautanod(antseledist *a, int i)
{ if (i<0)
     return NULL;
  while (a->ant)
        a=a->ant;
  while (i)
        { a=a->nx;
          i--;
        }
  return a;
}

// cautare arc - indiferent unde este pointerul in lista dublu inlantuita

arc* cautaarc( arc *a, int v1, int v2)
{ int aux;
  // daca pointerul arata spre elemente a caror prim varf este < v1
  // mergem spre dreapta
  while ( (a->nx) && (v1>a->v1) )
        a=a->nx;
  // daca pointerul arata spre elemente a caror prim varf este > v1
  // mergem spre stanga
  while ( (a->ant) && (v1<a->v1) )
        a=a->ant;
  // daca pointerul arata spre elemente a caror prim varf este v1
  // si v2 este > varful 2 din elementul curent parcurgem spre dreapta
  while ( (a->nx) && (v1==a->v1) && (v2>a->v2) )
        a=a->nx;
  // daca pointerul arata spre elemente a caror prim varf este v1
  // si v2 este < varful 2 din elementul curent parcurgem spre stanga
  while ( (a->ant) && (v1==a->v1) && (v2<a->v2) )
        a=a->ant;
  // daca elementul curent are varful1=v1 si varful2=v2
  // se returneaza adresa elementului curent
  if ( (a->v1==v1) && (a->v2==v2) )
     return a;
     else    // altfel se returneaza NULL echivalent cu inexistenta muchiei
         return NULL;
}

antseledist* minnesel(antseledist *d)
{ antseledist *min=new antseledist;
  int i=0;
  min->id=-1;
  while (d->ant)
        d=d->ant;
  while (d)
        { if ( (d->ptr->id) && ( (d->id<min->id) || (min->id==-1) ) && (d->id>0)  )
             min=d;
          d=d->nx;
          i++;
        }
  if (min->id<0)
     return NULL;
  return min;
}

void dijkstra(int noduri, arc *a, antseledist *d, antseledist *s, antseledist *ante)
{ int j,i,k;
  arc *aux2=NULL;
  antseledist *min=NULL,*aux1=NULL;
  for (j=0;j<noduri-1;j++)         // se parcurge nod cu nod
      { min=minnesel(d);           // se cauta nodul neselectat, aflat la distantza minima de nodul sursa (nu se verifica si nodurile aflate la infinit de sursa)
        if (min)                   // daca exista un astfel de nod (distantza intre 1 inclusiv si infinit exclusiv
           { min->ptr->id=0;       // se selecteaza
             k=index(min);         // se determina pozitia lui fie aceasta k in lista dinamica
           }
           else                    // daca nu exista nod pana la care de la sursa sa fie o distanta finita, algoritmul s-a terminat
               return;
        for (i=0;i<noduri;i++)      // se reactualizeaza (relaxeaza) costurile pana in acest moment
            { aux1=cautanod(d,i);   // pentru fiecare nod se cauta distantza de la sursa pana la el, oricat ar fi
              aux2=cautaarc(a,k,i); // se mai cauta (daca exista) arc intre nodul k si nodul curent i
              if (aux2)             // daca exista arc adica de la k la i NU este infinit
                 if ( (aux1->ptr->id) && ( ( aux1->id>min->id+aux2->cost ) || (aux1->id==-1) ) )  // daca nodul i este neselectat si distantza veche pana la i este mai mare decat distantza prin nodul k (a se avea in vedere ca distantza pana in nodul anterior selectat si arcul intre acesta si nodul curent nu pot fi infinit - cazuri eliminate anterior si de asemenea, din cauza notatiilor date valorii infinit=-1 sau -2, se verifica si cazul in care distanza veche nu este infinit=-1 (-2 nu ne intereseaza intrucat este distantza intre sursa si sursa))
                    { aux1->id=min->id+aux2->cost;     // se reactualizeaza distantza pana in nodul curent
                      adauganodsol(aux1->ptr->ptr,k,0);  // se pastraza solutzia...
                    }
                    else
                        if ( (aux1->ptr->id) && ( ( aux1->id==min->id+aux2->cost ) || (aux1->id==-1) ) )  // daca nodul i este neselectat si distantza veche pana la i este mai mare decat distantza prin nodul k (a se avea in vedere ca distantza pana in nodul anterior selectat si arcul intre acesta si nodul curent nu pot fi infinit - cazuri eliminate anterior si de asemenea, din cauza notatiilor date valorii infinit=-1 sau -2, se verifica si cazul in care distanza veche nu este infinit=-1 (-2 nu ne intereseaza intrucat este distantza intre sursa si sursa))
                             adauganodsol(aux1->ptr->ptr,k,1);  // se pastraza solutzia...

            }
      }
}

void initializari(int i, arc *a, antseledist *d, antseledist *s, antseledist *ante, int sursa)
{ arc *arcg;
  antseledist *aux;
  while (i>=0)                  // parcurgere noduri
        { aux=cautanod(d,i);    // caut distantza pana la nodul i
          if ( ( arcg=cautaarc(a,sursa,i) ) )   // caut arc intre sursa si nodul i
               aux->id=arcg->cost;              // daca este drum direct - in d[i] (ca lista dinamica)
             else                   // altfel
                 { if (i!=sursa)    // pentru nod curent diferit de sursa
                      aux->id=-1;   // distantza este infinit codificat ca -1
                      else              // pentru i=sursa
                          aux->id=0;    // distantza este 0
                 }
          if (aux->id>0)
             { aux->ptr->ptr->id=1;      // ante
               aux->ptr->ptr->ptr=new antseledist;  // new ante->ptr (primul elem predecesor)
               aux->ptr->ptr->ptr->ptr=aux->ptr->ptr->ptr->ant=aux->ptr->ptr;  // referiri la alti pointeri
               aux->ptr->ptr->ptr->nx=NULL;   // nu este urmator
               aux->ptr->ptr->ptr->id=sursa;  // informatia din ante->ptr
             }
          i--;          // se trece la nodul urmator
        }
  aux=cautanod(ante,sursa);     // ne pozitionam in lista nodurilor anterioare pe sursa
  aux->id=0;                   // anteriorul sursei este -2 (tot infinit), dar codofocat diferit pentru distingere de infinitul pentru care nu este drum direct
  aux=cautanod(s,sursa);        // ne pozitionam in lista nodurilor selectate pe sursa
  aux->id=0;                    // selectam sursa (0=selectat; 1=neselectat)
}

void dijkstar (int n,int sursa)
{
    int i,j,pas,k,min;
    // initializari
    for (i=1;i<=n;i++)
    {
       s[i]=0;
       distante[i]=cost[sursa][i];
        if (distante[i]<infinit)
                tata[i]=sursa;
        else
                tata[i]=0;
    }
    s[sursa]=1;
    tata[sursa]=0;
    distante[sursa]=0;
    for (pas=1;pas<=n-1;pas++)
    {
        // minimul distante
        min=infinit;
        int control=0;
        while (control==0)
        {    for (j=1;j<=n;j++)
                if (distante[j]<min && s[j]==0)
                    {
                    min=distante[j];
                    control=1;
                    k=j;
                    }
        }
       if (min==infinit) return;
       s[k]=1;
       for (j=1;j<=n;j++)
            if (s[j]==0 && distante[k]+cost[k][j]<distante[j])
            {
              distante[j]=distante[k]+cost[k][j];
              tata[j]=k;

            }

    }
}
