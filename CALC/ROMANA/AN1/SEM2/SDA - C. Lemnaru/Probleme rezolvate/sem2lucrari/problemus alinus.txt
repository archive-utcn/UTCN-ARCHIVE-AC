{
	printf("%d ",i);

	for(int j=1;j<=n;j++)
		if(matr[i][j]==1 && parc[j]==0)
			parcurgere_adancime(matr,parc,j,n);
}


void parcurgere_latime(int **matr,int *parc,int ic,int *sc,int *coada,int n)
{
	int k;
	if(ic<=*sc)
	{
		for(k=1;k<=n;k++)
			if(matr[coada[ic]][k]==1 && parc[k]==0)
			{
				parc[k]=1;
				(*sc)++;
				coada[*sc]=k;
			}
		parcurgere_latime(matr,parc,ic+1,&(*sc),coada,n);
	}
}

void Warshall(int **matr,int n)
{
	int i,j,k;
	for(k=1;k<=n;k++)
		for(i=1;i<=n;i++)
			for(j=1;j<=n;j++)
				if(matr[i][j]==0 && i!=k && j!=k)
					matr[i][j]=matr[i][k]*matr[k][j];
}

void descompunere(int **matr,int &nc,int n,int i,int *pus)
{

	nc++;
	pus[i]=1;

	printf("Componenta %d este formata din: %d ",nc,i);

	for(int j=1;j<=n;j++)
		if(matr[i][j]==1 && matr[j][i]==1 && i!=j && pus[j]==0)
		{
			pus[j]=1;
			printf("%d ",j);
		}
	printf("\n");

}

/sa se afle matricea drumurilor pt un graf orientat care e reprezentat prin liste de adiacenta(Roy-Warshall)
//in urma parcurgerii in adancime a grafului pornind de la un anumit varf (baza) vectorul s retine 1 pt toate nodurile la care s-a ajuns
//cu alte cuvinte vectorul s retine 1pt toate nodurile pt care exista drum de la nodul de pornire la ele.Astfel se obtine o linie a matricei.

#include<iostream.h>
#include<conio.h>
#include<stdlib.h>
#include<stdio.h>

struct t_nod
{
	int inf;
	t_nod *urm;
};


int parcurs[50],n,i,j,drumuri[50][50];
t_nod *lista[50];

void citire_graf(int &nrvf)
{
	char nume_fis[50]="D:\\fisiere\\fis6.txt";
	t_nod *p;

	//se creeaza si se deschide fisierul
	FILE *pf;
	pf=fopen(nume_fis,"r");

	//se citeste din fisier nr de varfuri
	fscanf(pf,"%d",&nrvf);

	//se citesc muchiile din fisier
	while(fscanf(pf,"%d%d",&i,&j)!=EOF)
	{
		p=(t_nod *)malloc(sizeof(t_nod));
		p->inf=j;
		p->urm=lista[i];
		lista[i]=p;
	}
	//se inchide fisierul
	fclose(pf);
}

void parcurgere_adancime(int baza,int nod)
{
	t_nod *p;
	if(baza!=nod)
		drumuri[baza][nod]=1;
	p=lista[nod];
	parcurs[nod]=1;
	while(p)
	{
		if(parcurs[p->inf]==0)
			parcurgere_adancime(baza,p->inf);
		else if(p->inf==baza)
      	drumuri[baza][baza]=1;
		p=p->urm;
	}
}

void main()
{
	clrscr();

	citire_graf(n);

	for(i=1;i<=n;i++)
	{
		for(j=1;j<=n;j++)
			parcurs[j]=0;
		parcurgere_adancime(i,i);
	}

	for(i=1;i<=n;i++)
	{
		for(j=1;j<=n;j++)
			cout<<drumuri[i][j]<<" ";
		cout<<endl;
	}

	getch();
}

void latime_nerecursiva(int ic,int *sc,int *parc,int *coada)
{
	lista *p;

	while(ic<=*sc)
	{
		p=prim[coada[ic]];
		while(p)
		{
			if(parc[p->info]==0)
			{
				parc[p->info]=1;
				(*sc)++;
				coada[*sc]=p->info;
			}
			p=p->urm;
		}
		ic++;
	}
}

void latime_recursiva(int ic,int *sc,int *parc,int *coada)
{
	lista *p;

	if(ic<=*sc)
	{
		p=prim[coada[ic]];
		while(p)
		{
			if(parc[p->info]==0)
			{
				parc[p->info]=1;
				(*sc)++;
				coada[*sc]=p->info;
			}
			p=p->urm;
		}
		latime_recursiva(ic+1,&(*sc),parc,coada);

	}
}

//Algoritmul lui Prim

#include <stdio.h>
#include <conio.h>

#define nmax 10
#define MAX 0x7fff

void prim(int n,int c[nmax][nmax],int muchii[nmax][2],int *cost)
{
	/*
		n -numarul nodurilor;
		c - matricea costurilor;
		muchii-muchiile arborelui de acoperire de cost minim;
		cost-costul arborelui de acoperire de cost minim
	*/
	int v[nmax];
	/*
		v[i]=0 daca i apartine arborelui;
		v[i]=j daca i nu apartine arborelui.
		j este nodul din arbore a.i.muchia (i,j) este de cost minim
	 */
	int i,j,k,minim;

	*cost=0;
	v[1]=0;
	for(i=2;i<=n;i++) v[i]=1;   //arborele este ({1},{})

	//determinarea celor n-1 muchii

	for(i=1;i<=n-1;i++)
	{
		//determinarea muchiei care se adauga arborelui

		minim=0x7fff;

		for(k=1;k<=n;k++)
			if(v[k]!=0)
				if(c[k][v[k]] < minim)
				{
					j=k;
					minim=c[k][v[k]];
				}
		muchii[i][0]=v[j];
		muchii[i][1]=j;
		*cost=*cost+c[j][v[j]];

		//reactualizare vector v

		v[j]=0;
		for(k=1;k<=n;k++)
			if(v[k]!=0)
				if(c[k][v[k]]>c[k][j])
					v[k]=j;
	}
}

{

	int n;//nr. nodurilor
	int c[nmax][nmax]; //matricea costurilor
	int muchii[nmax][2];//muchiile arborelui

	printf("\nNr. nodurilor=");
	scanf("%d",&n);

	for(i=1;i<=n;i++)
		for(j=1;j<=n;j++)
			 c[i][j]=MAX;

	//citirea costurilor(numere intregi)

	for(i=1;i<n;i++)
	{
		do
		{
			printf("\nIntroduceti 0 pentru terminare sau nodul adiacent\n");
			printf("	      cu  %2d > ca el =",i);
			scanf("%d",&j);
			if(j>0)
			{
				printf("\nCostul c[%d][%d]=",i,j);
				scanf("%d",&c[i][j]);
				c[j][i]=c[i][j]; //matricea este simetrica
			}
		}
		while(j>0);
	}

	prim(n,c,muchii,&cost);
	printf("\nCOSTUL ARBORELUI = %d",cost);

void DIJKSTRA(int **drum,int *tata,int *distante,int *stiva,int sursa,int n)
{
	//initializare date
	int i,j,pas,min,k;
	for(i=1;i<=n;i++)
	{
			distante[i]=drum[sursa][i];
			if(drum[sursa][i]<max)
				tata[i]=sursa;
	}

	stiva[sursa]=1;
	//distante[sursa]=0;
	//tata[sursa]=0;

	for(pas=1;pas<=n-1;pas++)
	{
		min=max;
		for(j=1;j<=n;j++)
			if(stiva[j]==0)
				if(min>distante[j])
				{
					min=distante[j];
					k=j;
				}

		if(min<max)
		{
			stiva[k]=1;

			for(j=1;j<=n;j++)
				if(stiva[j]==0)
					if(distante[j]>(distante[k]+drum[k][j]))
					{
						distante[j]=distante[k]+drum[k][j];
						tata[j]=k;
					}
		}
	}
}

void ROY_FLOYD(int **matr,int n)
{
	int i,j,k;
	for(k=1;k<=n;k++)
		for(i=1;i<=n;i++)
			for(j=1;j<=n;j++)
				if(matr[i][j]>matr[i][k]+matr[k][j])
					matr[i][j]=matr[i][k]+matr[k][j];

}

void drumulet(int **matr,int n,int i,int j)
{
	int k,gasit;

	k=1;
	gasit=0;

	while(k<=n && gasit==0)
	{
		if(i!=k && j!=k && (matr[i][k]+matr[k][j])==matr[i][j])
		{
			drumulet(matr,n,i,k);
			drumulet(matr,n,k,j);
			gasit=1;
		}
		k++;
	}
	if(gasit==0)
		printf("%d ",j);
}

		printf("Drumul de cost minim %d de la %d la %d este:\n",matr[i][j],i,j);
		printf("%d ",i);
		drumulet(matr,n,i,j);
	}
	else
		printf("Nu exista drum de la %d la %d",i,j);
}

int *bubble_sort(int *x,int n)
{
	int aux,sch,j=0;
	do
	{
		sch=0;
		j++;
		for(int i=0;i<n-j;i++)
			if(*(x+i)>*(x+i+1))
			{
				aux=*(x+i);
				*(x+i)=*(x+i+1);
				*(x+i+1)=aux;
				sch=1;
			}
    }
    while(sch);
    return x;

}
int *interschimbare(int *x,int n)
{
	int aux;
	for(int i=0;i<n-1;i++)
		for(int j=i+1;j<n;j++)
			if(*(x+i)>*(x+j))
			{
				aux=*(x+i);
				*(x+i)=*(x+j);
				*(x+j)=aux;
			}
	return x;
}
int *selectie(int *x,int n)
{
	int ind,aux;
	for(int i=0;i<n-1;i++)
	{
		ind=i;
		for(int j=i+1;j<n;j++)
			if(*(x+j)<*(x+ind))
				ind=j;
		if(i<ind)
		{
			aux=*(x+i);
			*(x+i)=*(x+ind);
			*(x+ind)=aux;
		}
	}
	return x;

}
int *insertie(int *x,int n)
{
	int ind,v;
	for(int i=1;i<n;i++)
	{
		ind=i-1;
		v=*(x+i);
		while(ind>=0 && (v<x[ind]))
		{
			x[ind+1]=x[ind];
			ind--;
		}
		x[ind+1]=v;
	}
	return x;
}
void interclasare(int *x,int *b,int st,int mijl,int dr)
{
	int i=st,j=mijl+1,k=st;
	while(i<=mijl && j<=dr)
	{
		if(x[i]<x[j])
			b[k++]=x[i++];
		else
			b[k++]=x[j++];
	}
	while(i<=mijl)
		b[k++]=x[i++];
	while(j<=dr)
		b[k++]=x[j++];
	for(i=st;i<=dr;i++)
		x[i]=b[i];
}

int *div_et_impera(int *x,int *b,int st,int dr)
{
	int mijl;
	if(st<dr)
	{
		mijl=(st+dr)/2;
		div_et_impera(x,b,st,mijl);
		div_et_impera(x,b,mijl+1,dr);
		interclasare(x,b,st,mijl,dr);
	}
	return x;
}

int *quick_sort(int *x,int st,int dr)
{
	int i,j,pivot,a;
	i=st;
	j=dr;
	pivot=x[(st+dr)/2];
	do
	{
		while(x[i]<pivot)
			i++;
		while(x[j]>pivot)
			j--;
		if(i<=j)
		{
			a=x[i];
			x[i]=x[j];
			x[j]=a;
			i++;
			j--;
		}
	}
	while(i<=j);
	if(st<j)
		quick_sort(x,st,j);
	if(i<dr)
		quick_sort(x,i,dr);
	return x;
}

				b1[j]+=++inc;
	}
	for(i=0;i<n;i++)

	}
	return xx;
}

#include <stdio.h>

#define  nmax 10

typedef struct tip_nod

void prod_matr(int n,long c[nmax][nmax],int d[nmax+1])
{
	int i,j,k,p,poz;
	long min,val;

	for(i=1;i<=n;i++)
		c[i][i]=0;
	for(p=1;p<=n-1;p++)
		for(i=1;i<=n-p;i++)
		{
			j=i+p;
			min=0x7fffffff;
			for(k=i;k<=j-1;k++)
			{
				val=c[i][k]+c[k+1][j]+(long)d[i]*d[k+1]*d[j+1];
				if(val<min)
				{
					min=val;
					poz=k;
				}
			}
			c[i][j]=min;
			c[j][i]=poz;
		}
}

{

	p=(TIP_NOD *)malloc(sizeof(TIP_NOD));
	p->ind1=i;
	p->ind2=j;

	if(i<j)
	{
		p->stg=constr_arbore(p->stg,i,c[j][i],c);
		p->dr=constr_arbore(p->dr,c[j][i]+1,j,c);
	}
	else
	{
		p->stg=0;
		p->dr=0;
	}

	return p;
}

void subsir_dinamic()
{

	int i,j,max,poz_max;

	l[n]=1;
	poz[n]=-1;

	for(i=n-1;i>0;i--)
	{

		l[i]=1;
		poz[i]=-1;
		for(j=i+1;j<=n;j++)
			if(a[i]<=a[j] && l[i]<=1+l[j])
			{

				l[i]=l[j]+1;
				poz[i]=j;

			}
	}

	//afisarea solutiei
	max=l[1];
	poz_max=1;
	for(i=2;i<=n;i++)
		if(max<l[i])

			max=l[i];
			poz_max=i;
		}
	cout<<"Lungimea celui mai lung subsir crescator este:\n"<<max;
	cout<<"\nCel mai lung subsir crescator este:\n";
	for(i=poz_max;i!=-1;i=poz[i])
		cout<<a[i]<<" ";

}

*

	Problema comis voiajorului: Se da un graf neorientat G =(X,Y) in care toate nodurile sunt unite
	intre ele printr-o muchie, careia i se asociaza un cost strict pozitiv. Se cere determinarea unui ciclu,
	care sa inceapa dintr-un nod i, sa treaca exact o data prin toate nodurile si sa se intoarca in nodul initial.
	Se cere ca ciclul gasit sa aiba un cost minim.

*/

#include <stdio.h>
#include <conio.h>
#define n_max 10

void comis_voiajor(int n,int c[n_max][n_max],int i,int ciclu[n_max+1],int *cost)
{
	/*
		n este nr.nodurilor;
		c este matricea costurilor;
		i este nodul de start;
		cost este costul ciclului
	*/

	int p[n_max];
	int k,v,j,vmin,costmin;

	for(k=1;k<=n;k++)
		p[k]=0;

	*cost=0;

	p[i]=1;
	ciclu[1]=i;
	v=i; /*nodul curent */

	for(k=1;k<n;k++) /* se adauga pe rand n-1 muchii */
	{

		costmin=0x7fff;

		//gasirea muchiei de cost minim care are nodul de origine v

		for(j=1;j<=n;j++)
			if((p[j]==0)&&(c[v][j]<costmin))
			{
				costmin=c[v][j];
				vmin=j;
			}

		*cost=*cost+costmin;
		ciclu[k+1]=vmin;
		p[vmin]=1;
		v=vmin;
	}

	ciclu[n+1]=i;
	*cost=*cost+c[v][i];

}

void main()
{

	clrscr();

	int i,j,n,cost_ciclu,ciclu[n_max],c[n_max][n_max];

	printf("\nNr.nodurilor grafului = ");

	for(i=1;i<=n;i++)
		for(j=1;j<=n;j++)
			c[i][j]=0x7fff;

	printf("\nIntroduceti costurile muchiilor:");
	for(i=1;i<=n-1;i++)
	{
		while(1)
		{

			printf("Nodul adiacent lui %d =",i);
			scanf("%d",&j);
			if(j!=0)
			{
				printf("Costul muchiei (%d,%d)=",i,j);
				scanf("%d",&c[i][j]);
				c[j][i]=c[i][j];
			}
			else break;

		 }
	}

	i=1;

	comis_voiajor(n,c,i,ciclu,&cost_ciclu);

	printf("\nCOSTUL CICLULUI =%d\n",cost_ciclu);
	printf("\nCICLUL=");
	for(i=1;i<=n+1;i++)
		printf("%3d",ciclu[i]);

	getch();

}
