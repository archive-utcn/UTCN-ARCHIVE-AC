#include <stdio.h>
#include <stdlib.h>
int viz[50], viz2[50], a[50][50],n;
FILE *f;

void BFS(int nod)
{
    int pi, ps, v[100],i;
    pi=1;
    ps=1;
    v[pi]=nod=1;

    while(pi<=ps)
    {
        for(i=1;i<=n;i++)
         if ((viz[i]==0) && (a[nod][i]==1))
         {
             ps++;
             viz[i]=1;
             printf("%d",i);
             v[ps]=i;
         }
         pi++;
         nod=v[pi];
    }
}

void DFS(int nod)
{
    int i;
    viz2[nod]=1;
    printf("%d", nod);
    for(i=1;i<=n;i++)
    if((viz2[i]==0) && (a[nod][i]==1))
    DFS(i);
}

void citire()
{
    int i, j;
    f=fopen("fis.txt", "r");
     fscanf(f,"%d", &n);
     for(i=1;i<=n;i++)
     {
         for(j=1;j<=n;j++)
         {
             fscanf(f,"%d", &a[i][j]);
             printf("%d", a[i][j]);
         }
         printf("\n");
     }
     fclose(f);
}


int main()
{

    citire();
    printf("\n");
    BFS(1);
    printf("\n");
    DFS(1);

    return 0;
}
