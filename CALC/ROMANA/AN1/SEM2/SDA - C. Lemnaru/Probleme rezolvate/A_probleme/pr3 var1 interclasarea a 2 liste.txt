#include <stdio.h>
#include <conio.h>
#include <stdlib.h>
#include <string.h>

typedef struct cuvant
{
    int fr;
    char cuv[20];
    struct cuvant *urm,*prec;
} NOD;

void creare(NOD **prim,NOD **ultim, char cuv[])
{
    NOD *p=*prim;
    int inserat=0, gasit=0;
    while ((p!=0) && (inserat==0) && (gasit==0))
    {
        int rez;
        rez=strcmp(cuv,p->cuv);
        if (rez==0)
        {
            p->fr++;
            inserat=1;
        }
        else if (rez>0) p=p->urm;
             else gasit=1;
    }

    if ((p==0)||(gasit==1))
    {
        NOD *q;
        q=(NOD*)malloc(sizeof(NOD));
        strcpy(q->cuv,cuv);
        q->fr=1;
        q->urm=q->prec=0;
        if (p==0)
        {
            if (*ultim==0)        //lista este vida
            {
                *prim=*ultim=q;
                (*prim)->prec=0;
                (*ultim)->urm=0;
            }
            else                  //lista nu e vida
            {
                (*ultim)->urm=q;
                q->prec=*ultim;
                *ultim=q;
                (*ultim)->urm=0;
            }
        }
    if (gasit==1)
    {
        q->prec=p->prec;
        q->urm=p;
        if (p->prec!=0)
        {
        p->prec->urm=q;
        p->prec=q;
        }
        if (p==*prim)
        {
          *prim=q;
          (*prim)->urm=p;
           p->prec=*prim;
        }
     }
    }
}

void afisare1(NOD *prim)
{
    printf("\n\nAfisare in ordine alfabetica:\n\n");
    NOD *p=prim;
    while (p!=0)
    {
        printf("\n%s",p->cuv);
        p=p->urm;
    }
}

void afisare2(NOD *ultim)
{
    printf("\n\nAfisare in ordine alfabetica inversa:\n\n");
    NOD *p=ultim;
    while (p!=0)
    {
        printf("\n%s",p->cuv);
        p=p->prec;
    }
}

void main()
{
