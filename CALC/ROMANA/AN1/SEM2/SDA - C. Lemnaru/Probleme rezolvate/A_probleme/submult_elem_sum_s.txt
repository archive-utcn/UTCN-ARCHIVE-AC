/*3.9.Se considerã o mulþime formatã din n elemente numere întregi. Sã se genereze toate submulþimile
acestei mulþimi având proprietatea cã suma elementelor lor este egalã cu S.
*/
#include <stdio.h>
#include <stdlib.h>
#define n 3
int S=3;
int A[3];
int x[3];
void afisare(int x[],int k)
{
    int i;
    for(i=0;i<k;i++)
    printf("%d",x[i]);

}
int cond(int x[],int k)
{
    int i;
    int sum=0;
    for(i=0;i<k;i++)
    sum=sum+x[i];
    if (sum<=S) return 1;
    return 0;
}
int sum(int x[],int k)
{
    int i;
    int sum=0;
    for(i=0;i<k;i++)
    sum=sum + x[i];
    return sum;
}
void backtracking(int k)
{
    int i;
    for(i=0;i<3;i++)
    {
    x[k]=A[i];
        if(cond(x,k)!=0)
        {
            if(sum(x,k)==S)
            {
                afisare(x,k);
                printf("\n");
            }
            else backtracking(k+1);
        }

    }



}


int main()
{
    int i;
    for(i=0;i<3;i++)
    scanf("%d",&A[i]);
    for(i=0;i<3;i++)
    printf("%d",A[i]);
    printf("\n");
    backtracking(0);


    return 0;
}
