#include <stdio.h>
#include <stdlib.h>
#define M 11

typedef struct student
{
    char *cheie;
    char nume_prenume[50];
    char facultate[30];
    int an;
    struct student *urm;
}Nod;

Nod *HT[M];

int f(char *cheie)
{
    int i,a;
    a=0;
    for(i=0;i<lenght(cheie);i++)
        a=a+*(cheie+i);
    return (a % M);
}

void elib_nod(Nod *p)
{
    free(p);
}

Nod *cautare(Nod *ht[],char n_p[])
{
    int f(); int a;
    int h,gasit=0;
    Nod *p;
    h=f(n_p);
    p=ht[h];
    while((p!=0) && (gasit!=0))
    {
        if(strcmp(p->nume_prenume,n_p)==0)
          {
              gasit==1;
          }
    }
    if(gasit==1)
    {
        printf("\nEROARE! Studentul %s deja exista!\n",n_p);
        p=NULL;
    }
    else
    {
        p=(Nod *)malloc(sizeof(Nod));
        strcpy(p->nume_prenume,n_p);
        p->urm=ht[h];
        ht[h]=p;
    }
    return p;
}

Nod *adaugare()
{
    Nod *p,*q;
    int h;
    p=(Nod *)malloc(sizeof(Nod));
    if(p==0)
    {
        printf("\n Memorie insuficienta!\n");
        exit(1);
    }
    if(citire_nod(p)!=1)
    {
        printf("\n Eroare citire!\n");
        elib_nod(p);
        return 0;
    }
    h=f(p->cheie);
    if(HT[h]==0)
    {
        HT[h]=p;
        p->urm=0;
        return p;
    }
    else
    {
        q=cautare(p->cheie);
        if(q==0)
        {
            p->urm=HT[h];
            HT[h]=p;
            return p;
        }
        else
        {
            printf("\nExista o inregistrare cu cheia respectiva.\n");
            printf(p,q);
        }
    }
}

int main()
{
    int i;
    for(i=0;i<=M-1;i++)
        HT[i]=0;
    return 0;
}
