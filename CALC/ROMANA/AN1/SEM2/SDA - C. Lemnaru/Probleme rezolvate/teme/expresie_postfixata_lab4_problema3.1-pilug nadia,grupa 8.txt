#include <stdio.h>

typedef struct stiv
{
    char c;
    struct stiv *urm;
}STIVA;

typedef struct tip_nod
{
    char ch;
    struct tip_nod *stg,*dr;
}TIP_NOD;
FILE *f;
void inserare (STIVA **prim, STIVA **ultim,char key)
{
    STIVA *p;
    if (*prim==0)
    {
        p=(STIVA*)malloc(sizeof(STIVA));
        p->c=key;
        *prim=p;
        *ultim=p;
        (*ultim)->urm=0;
    }
    else
    {
        p->c=key;
        p->urm=*prim;
        *prim=p;
    }
}

void elim (STIVA **prim, STIVA **ultim)
{
    STIVA *p;
    if(*prim==0)
    {
        //lista e vida
        return;
    }
    p=*prim;
    *prim=(*prim)->urm;
    free(p);
    if(*prim==0)
        *ultim=0;
    //lista a devenit vida dupa stergerea unicului nod
}

            //stiva:last in,first out

            elim(prim,ultim);
            //ultimul a devenit astfel pen-ultimul
            p->dr->ch=ultim->c;
            elim(prim,ultim);
        }
    }
}
