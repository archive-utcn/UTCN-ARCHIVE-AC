   if (strcmp(p->gen,inreg->gen)==0) { while (strcmp(p->nume,inreg->nume)<0) p=p->urm; } //au acelasi gen si comparam dupa nume
   else
   {
       //inserarea inaintea nodului p
       n=sizeof(nod);

       p->urm=q;
       if(p==ultim) ultim=q;
   }
 }
