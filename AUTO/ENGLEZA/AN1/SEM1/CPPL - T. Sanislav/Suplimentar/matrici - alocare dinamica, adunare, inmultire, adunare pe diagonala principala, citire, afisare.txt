#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define M_DIM 2
#define N_DIM 2
#define P_DIM 2

int  **alloc_matrix(int, int);
void free_matrix(int**, int);
int  read_matrix(int**, int, int, const char*);
void print_matrix(int**, int, int, const char*);
int  **multiply_matrix(int**, int**, int, int, int);
int  **add_matrix(int**, int**, int, int);
void add_column(int**, int, int);

int main(){
    int **matrix1, **matrix2, **result;

    //MULTIPLY
    matrix1 = alloc_matrix(M_DIM, N_DIM);
    matrix2 = alloc_matrix(N_DIM, P_DIM);

    read_matrix(matrix1, M_DIM, N_DIM, "First matrix");
    print_matrix(matrix1, M_DIM, N_DIM, "First matrix");
    read_matrix(matrix2, N_DIM, P_DIM, "Second matrix");
    print_matrix(matrix2, N_DIM, P_DIM, "Second matrix");

    result = multiply_matrix(matrix1, matrix2, M_DIM, N_DIM, P_DIM);
    print_matrix(result, M_DIM, P_DIM, "Product");

    free_matrix(result, N_DIM);
    free_matrix(matrix1, M_DIM);
    free_matrix(matrix2, N_DIM);

    //ADD
    matrix1 = alloc_matrix(M_DIM, N_DIM);
    matrix2 = alloc_matrix(M_DIM, N_DIM);

    read_matrix(matrix1, M_DIM, N_DIM, "First matrix");
    print_matrix(matrix1, M_DIM, N_DIM, "First matrix");
    read_matrix(matrix2, N_DIM, P_DIM, "Second matrix");
    print_matrix(matrix2, N_DIM, P_DIM, "Second matrix");

    result = add_matrix(matrix1, matrix2, M_DIM, N_DIM);
    print_matrix(result, M_DIM, N_DIM, "Sum");

    free_matrix(result, M_DIM);
    free_matrix(matrix1, M_DIM);
    
    free_matrix(matrix2, M_DIM);
    //col-sum coloana principala
    matrix1 = alloc_matrix(M_DIM, N_DIM);

    read_matrix(matrix1, M_DIM, N_DIM, "First matrix");
    print_matrix(matrix1, M_DIM, N_DIM, "First matrix");

    add_column(matrix1, M_DIM, N_DIM);

    free_matrix(matrix1, M_DIM);

    return 0;
}
int **alloc_matrix(int m, int n) //m linii, n coloane
{
    int **matrix = NULL;
    int i = 0;

    matrix =  (int **) malloc(n * sizeof(int)); //aloacam spatiu pt toata matricea[][]
    for ( i = 0; i < m; i++)
        *(matrix + i) = (int *) malloc (n * sizeof(int)); //alocam spatiu pt fiecare element din mat []

    return matrix;
}

void free_matrix(int **matrix, int m)
{
    int i = 0;

    for ( i = 0; i < m; i++)
        free(*(matrix + i));  //eliberam memoria alocata pt fiecare element din matrice

    free(matrix); //eliberam mem alocata pt matrix
}

int read_matrix(int **matrix, int m, int n, const char *name)
{
    int i = 0, j = 0;

    for (i = 0; i < m; i ++)
        for (j = 0; j < n; j++)
    {
        printf("%s[%d][%d] = ", name, i , j);
        scanf("%d", *(matrix + i) + j);
    }

    return 1;
}

void print_matrix(int **matrix, int m, int n, const char *name)
{
    int i = 0, j = 0;

    printf("Matrix %s is: \n---------------\n", name);
    for (i = 0; i < m; i++)
    {
        for (j = 0; j < n; j++)
        {
          printf("%d ", *(*(matrix + i) + j));
        }
        printf("\n");
    }
    printf("------------------\n");
}

int **multiply_matrix(int **matrix1, int **matrix2, int m, int n, int p)
{
    int i = 0, j = 0, k = 0;
    int **result = alloc_matrix(m, p);  //allocam matricea pt rezultat

    for (i = 0; i < m; i++)
        for (j = 0; j < p; j++)
            *(*(result + i) + j) = 0;  //setam fiecare element din matricea proaspat creata la 0

    for (i = 0; i < m; i++)
        for (j = 0; j < n; j++)
            for (k = 0; k < p; k++)
                *(*(result + i) +j) += (*(*(matrix1 + i) + j)) * (*(*(matrix2 + j) + k));

    return result;
}

int **add_matrix(int **matrix1, int **matrix2, int m, int n)
{
    int i = 0, j = 0;
    int **result = alloc_matrix(m, n);

    for (i = 0; i < m; i++)
        for (j = 0; j < n; j++)
            *(*(result + i) + n) = 0;  //setam fiecare element din matricea proaspat creata la 0

    for (i = 0; i < m; i++)
        for (j = 0; j < n; j++)
            *(*(result + i) + j) = (*(*(matrix1 + i) + j)) + (*(*(matrix2 + i) + j)); //pt scadere se pune - intre cei doi membri

    return result;
}

void add_column(int **matrix, int m, int n)
{
    int i = 0, j = 0, sum = 0;

    for (i = 0; i < m; i++)
        for (j = 0; j < n; j++)
            if (i == j)
            sum += *(*(matrix + i) + j);

    printf("The sum of the elements located on the principal column of the matrix is: %d \n", sum);
}
